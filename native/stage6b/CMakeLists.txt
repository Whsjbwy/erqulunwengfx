cmake_minimum_required(VERSION 3.20)
project(auv_stage6b LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ompl 2.0.2 EXACT CONFIG REQUIRED)
find_path(STAGE6B_JSON_INCLUDE nlohmann/json.hpp REQUIRED)

add_library(auv6b_model STATIC model.cpp)
target_include_directories(auv6b_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(auv6b_model_tests model_tests.cpp)
target_link_libraries(auv6b_model_tests PRIVATE auv6b_model)
add_executable(auv6b_runner ompl_runner.cpp)
target_include_directories(auv6b_runner SYSTEM PRIVATE ${STAGE6B_JSON_INCLUDE})
target_link_libraries(auv6b_runner PRIVATE auv6b_model ompl::ompl)

foreach(target auv6b_model auv6b_model_tests auv6b_runner)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(MINGW)
    target_link_options(auv6b_runner PRIVATE -municode)
endif()
enable_testing()
add_test(NAME frozen_model_geometry COMMAND auv6b_model_tests)
