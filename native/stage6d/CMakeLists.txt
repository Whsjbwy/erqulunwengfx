cmake_minimum_required(VERSION 3.20)
project(auv_stage6d_diagnostic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ompl 2.0.2 EXACT CONFIG REQUIRED)
find_path(STAGE6D_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
add_executable(auv6d_probe interface_probe.cpp ../stage6b/model.cpp)
target_include_directories(auv6d_probe PRIVATE ../stage6b ../stage6c)
target_include_directories(auv6d_probe SYSTEM PRIVATE ${STAGE6D_JSON_INCLUDE})
target_link_libraries(auv6d_probe PRIVATE ompl::ompl)
target_compile_options(auv6d_probe PRIVATE -Wall -Wextra -Wpedantic)
if(MINGW)
    target_link_options(auv6d_probe PRIVATE -municode)
endif()
enable_testing()
# Passing this diagnostic test detects a blocked contract, not a valid model gate.
add_test(NAME detect_frozen_pitch_contract_gap COMMAND auv6d_probe --self-test)
