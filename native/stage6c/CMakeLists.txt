cmake_minimum_required(VERSION 3.20)
project(auv_stage6c LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ompl 2.0.2 EXACT CONFIG REQUIRED)
find_path(STAGE6C_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
add_library(auv6c_frozen_model STATIC ../stage6b/model.cpp)
target_include_directories(auv6c_frozen_model PUBLIC ../stage6b)
add_executable(auv6c_runner ompl_runner.cpp)
add_executable(auv6c_adapter_tests adapter_tests.cpp)
add_executable(auv6c_model_tests ../stage6b/model_tests.cpp)
foreach(target auv6c_runner auv6c_adapter_tests auv6c_model_tests)
    target_include_directories(${target} SYSTEM PRIVATE ${STAGE6C_JSON_INCLUDE})
    target_link_libraries(${target} PRIVATE auv6c_frozen_model ompl::ompl)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(MINGW)
    target_link_options(auv6c_runner PRIVATE -municode)
    target_link_libraries(auv6c_runner PRIVATE psapi)
endif()
enable_testing()
add_test(NAME frozen_model_geometry COMMAND auv6c_model_tests)
add_test(NAME official_adapter_metric COMMAND auv6c_adapter_tests)
