cmake_minimum_required(VERSION 3.20)
project(auv_stage6dr_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ompl 2.0.2 EXACT CONFIG REQUIRED)
find_path(STAGE6DR_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
add_library(auv6dr_frozen_model STATIC ../stage6b/model.cpp)
target_include_directories(auv6dr_frozen_model PUBLIC ../stage6b)
add_executable(auv6dr_runner runner.cpp)
add_executable(auv6dr_contract_tests contract_tests.cpp)
add_executable(auv6dr_original_probe ../stage6d/interface_probe.cpp)
add_executable(auv6dr_model_tests ../stage6b/model_tests.cpp)
add_executable(auv6dr_frozen_adapter_tests ../stage6c/adapter_tests.cpp)
foreach(target auv6dr_runner auv6dr_contract_tests auv6dr_original_probe auv6dr_model_tests auv6dr_frozen_adapter_tests)
    target_include_directories(${target} PRIVATE ../stage6c)
    target_include_directories(${target} SYSTEM PRIVATE ${STAGE6DR_JSON_INCLUDE})
    target_link_libraries(${target} PRIVATE auv6dr_frozen_model ompl::ompl)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
if(MINGW)
    target_link_options(auv6dr_runner PRIVATE -municode)
    target_link_options(auv6dr_original_probe PRIVATE -municode)
endif()
enable_testing()
add_test(NAME frozen_model_geometry COMMAND auv6dr_model_tests)
add_test(NAME frozen_adapter_metric COMMAND auv6dr_frozen_adapter_tests)
add_test(NAME original_failure_remains_detected COMMAND auv6dr_original_probe --self-test)
add_test(NAME repaired_pitch_contract COMMAND auv6dr_contract_tests)
